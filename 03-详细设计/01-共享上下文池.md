# 共享上下文池 (SharedContext) 详细设计

## 1. 概述

### 1.1 设计目标

共享上下文池是多Agent协作的核心组件，负责在多个Agent之间共享信息，确保生成内容的一致性和连贯性。

### 1.2 核心功能

```
│      - 存储标准术语定义                                     │
│      - 记录术语使用次数                                     │
│      - 支持术语别名                                         │
```

## 2. 数据结构

### 2.1 核心数据模型

```typescript
interface SharedContextData {
  // 项目信息
  projectOverview: string;
  technicalRequirements: string[];
  scoringCriteria: ScoringCategory[];
  
  // 术语表
  terminology: Map<string, TermEntry>;
  
  // 章节摘要
  chapterSummaries: Map<string, ChapterSummary>;
  
  // 风格指南
  styleGuide: StyleGuide;
  
  // 交叉引用
  references: Reference[];
  
  // 评分点覆盖
  requirementCoverage: Map<string, RequirementCoverage>;
  
  // 元数据
  metadata: {
    created_at: Date;
    updated_at: Date;
    version: number;
  };
}

interface TermEntry {
  term: string;
  definition: string;
  aliases: string[];
  usage_count: number;
  first_used_in: string;  // 首次使用的章节ID
  created_at: Date;
}

interface ChapterSummary {
  chapter_id: string;
  summary: string;        // 200字以内
  key_points: string[];
  word_count: number;
  generated_at: Date;
}

interface StyleGuide {
  tone: 'formal' | 'professional' | 'technical';
  perspective: 'first_person_plural' | 'third_person';
  formatting_rules: FormattingRule[];
  forbidden_words: string[];
  preferred_phrases: Map<string, string>;
}

interface FormattingRule {
  type: 'heading' | 'list' | 'table' | 'paragraph';
  rule: string;
  example: string;
}

interface Reference {
  id: string;
  from_chapter_id: string;
  to_chapter_id: string;
  context: string;
  type: 'see_also' | 'as_shown_in' | 'refer_to' | 'mentioned_in';
  created_at: Date;
}

interface RequirementCoverage {
  requirement_id: string;
  requirement_desc: string;
  covered_by: string[];      // 覆盖的章节ID列表
  coverage_status: 'full' | 'partial' | 'none';
  last_updated: Date;
}
```

## 5. 实现代码

```typescript
import { EventEmitter } from 'events';
import { ReadWriteLock } from '../utils/ReadWriteLock';
import { PersistenceService } from '../services/PersistenceService';
import { Logger } from '../utils/Logger';

export class SharedContext extends EventEmitter implements ISharedContext {
  private data: SharedContextData;
  private lock: ReadWriteLock;
  private persistence: PersistenceService;
  private logger: Logger;
  private bidId: string;
  
  constructor(
    persistence: PersistenceService,
    logger: Logger
  ) {
    super();
    this.persistence = persistence;
    this.logger = logger;
    this.lock = new ReadWriteLock();
    this.data = this.createEmptyContext();
  }
  
  async initialize(projectInfo: ProjectInfo): Promise<void> {
    await this.lock.acquireWrite();
    try {
      this.data = {
        ...this.data,
        projectOverview: projectInfo.projectOverview,
        technicalRequirements: projectInfo.technicalRequirements,
        scoringCriteria: projectInfo.scoringCriteria,
        metadata: {
          ...this.data.metadata,
          created_at: new Date(),
          updated_at: new Date()
        }
      };
      
      this.emit('initialized', this.data);
      this.logger.info('共享上下文已初始化');
    } finally {
      this.lock.releaseWrite();
    }
  }
  
  async getSnapshot(): Promise<SharedContextData> {
    await this.lock.acquireRead();
    try {
      return this.deepClone(this.data);
    } finally {
      this.lock.releaseRead();
    }
  }
  
  async addTerm(term: TermEntry): Promise<void> {
    await this.lock.acquireWrite();
    try {
      const existing = this.data.terminology.get(term.term);
      
      if (existing) {
        // 更新现有术语
        existing.usage_count += 1;
        existing.aliases = [...new Set([...existing.aliases, ...term.aliases])];
      } else {
        // 添加新术语
        this.data.terminology.set(term.term, {
          ...term,
          usage_count: 1,
          created_at: new Date()
        });
      }
      
      this.data.metadata.updated_at = new Date();
      this.emit('term_added', term);
    } finally {
      this.lock.releaseWrite();
    }
  }
  
  getTerm(term: string): TermEntry | undefined {
    return this.data.terminology.get(term);
  }
  
  getAllTerms(): TermEntry[] {
    return Array.from(this.data.terminology.values());
  }
  
  async updateTermUsage(term: string): Promise<void> {
    await this.lock.acquireWrite();
    try {
      const entry = this.data.terminology.get(term);
      if (entry) {
        entry.usage_count += 1;
        this.data.metadata.updated_at = new Date();
      }
    } finally {
      this.lock.releaseWrite();
    }
  }
  
  async updateChapterSummary(chapterId: string, summary: ChapterSummary): Promise<void> {
    await this.lock.acquireWrite();
    try {
      this.data.chapterSummaries.set(chapterId, summary);
      this.data.metadata.updated_at = new Date();
      
      this.emit('summary_updated', { chapterId, summary });
      this.logger.debug(`章节摘要已更新: ${chapterId}`);
    } finally {
      this.lock.releaseWrite();
    }
  }
  
  getChapterSummary(chapterId: string): ChapterSummary | undefined {
    return this.data.chapterSummaries.get(chapterId);
  }
  
  getRelatedSummaries(chapterId: string): ChapterSummary[] {
    const summaries: ChapterSummary[] = [];
    
    // 获取父章节摘要
    const parentId = this.getParentChapterId(chapterId);
    if (parentId) {
      const parentSummary = this.data.chapterSummaries.get(parentId);
      if (parentSummary) {
        summaries.push(parentSummary);
      }
    }
    
    // 获取同级章节摘要
    const siblings = this.getSiblingChapterIds(chapterId);
    for (const siblingId of siblings) {
      const siblingSummary = this.data.chapterSummaries.get(siblingId);
      if (siblingSummary) {
        summaries.push(siblingSummary);
      }
    }
    
    return summaries;
  }
  
  async updateStyleGuide(guide: StyleGuide): Promise<void> {
    await this.lock.acquireWrite();
    try {
      this.data.styleGuide = guide;
      this.data.metadata.updated_at = new Date();
      
      this.emit('style_guide_updated', guide);
    } finally {
      this.lock.releaseWrite();
    }
  }
  
  getStyleGuide(): StyleGuide {
    return this.data.styleGuide;
  }
  
  async addReference(ref: Reference): Promise<void> {
    await this.lock.acquireWrite();
    try {
      // 检查是否已存在相同引用
      const exists = this.data.references.some(
        r => r.from_chapter_id === ref.from_chapter_id && 
             r.to_chapter_id === ref.to_chapter_id
      );
      
      if (!exists) {
        this.data.references.push({
          ...ref,
          id: this.generateId(),
          created_at: new Date()
        });
        
        this.data.metadata.updated_at = new Date();
        this.emit('reference_added', ref);
      }
    } finally {
      this.lock.releaseWrite();
    }
  }
  
  getReferencesFromChapter(chapterId: string): Reference[] {
    return this.data.references.filter(r => r.from_chapter_id === chapterId);
  }
  
  getReferencesToChapter(chapterId: string): Reference[] {
    return this.data.references.filter(r => r.to_chapter_id === chapterId);
  }
  
  async updateRequirementCoverage(requirementId: string, chapterId: string): Promise<void> {
    await this.lock.acquireWrite();
    try {
      let coverage = this.data.requirementCoverage.get(requirementId);
      
      if (!coverage) {
        coverage = {
          requirement_id: requirementId,
          requirement_desc: this.getRequirementDesc(requirementId),
          covered_by: [],
          coverage_status: 'none',
          last_updated: new Date()
        };
        this.data.requirementCoverage.set(requirementId, coverage);
      }
      
      if (!coverage.covered_by.includes(chapterId)) {
        coverage.covered_by.push(chapterId);
        coverage.coverage_status = 'full';
        coverage.last_updated = new Date();
        
        this.data.metadata.updated_at = new Date();
        this.emit('coverage_updated', { requirementId, chapterId });
      }
    } finally {
      this.lock.releaseWrite();
    }
  }
  
  getCoverageStatus(): RequirementCoverage[] {
    return Array.from(this.data.requirementCoverage.values());
  }
  
  getUncoveredRequirements(): string[] {
    const uncovered: string[] = [];
    
    for (const [id, coverage] of this.data.requirementCoverage) {
      if (coverage.coverage_status === 'none') {
        uncovered.push(id);
      }
    }
    
    return uncovered;
  }
  
  async batchUpdate(updates: ContextUpdate[]): Promise<void> {
    await this.lock.acquireWrite();
    try {
      for (const update of updates) {
        switch (update.type) {
          case 'term':
            await this.addTerm(update.data);
            break;
          case 'summary':
            await this.updateChapterSummary(update.data.chapter_id, update.data);
            break;
          case 'reference':
            await this.addReference(update.data);
            break;
          case 'coverage':
            await this.updateRequirementCoverage(update.data.requirement_id, update.data.chapter_id);
            break;
        }
      }
    } finally {
      this.lock.releaseWrite();
    }
  }
  
  async save(): Promise<void> {
    await this.lock.acquireRead();
    try {
      await this.persistence.saveContext(this.bidId, this.data);
      this.logger.info('共享上下文已保存');
    } finally {
      this.lock.releaseRead();
    }
  }
  
  async load(bidId: string): Promise<void> {
    await this.lock.acquireWrite();
    try {
      this.bidId = bidId;
      const loaded = await this.persistence.loadContext(bidId);
      
      if (loaded) {
        this.data = loaded;
        this.logger.info('共享上下文已加载');
      } else {
        this.data = this.createEmptyContext();
        this.logger.info('创建新的共享上下文');
      }
    } finally {
      this.lock.releaseWrite();
    }
  }
  
  private createEmptyContext(): SharedContextData {
    return {
      projectOverview: '',
      technicalRequirements: [],
      scoringCriteria: [],
      terminology: new Map(),
      chapterSummaries: new Map(),
      styleGuide: {
        tone: 'professional',
        perspective: 'first_person_plural',
        formatting_rules: [],
        forbidden_words: [],
        preferred_phrases: new Map()
      },
      references: [],
      requirementCoverage: new Map(),
      metadata: {
        created_at: new Date(),
        updated_at: new Date(),
        version: 1
      }
    };
  }
  
  private deepClone<T>(obj: T): T {
    return JSON.parse(JSON.stringify(obj, (key, value) => {
      if (value instanceof Map) {
        return Object.fromEntries(value);
      }
      return value;
    }));
  }
}
```

## 6. 并发控制

### 6.1 读写锁实现

```typescript
class ReadWriteLock {
  private readers: number = 0;
  private writer: boolean = false;
  private waitingWriters: Function[] = [];
  
  async acquireRead(): Promise<void> {
    while (this.writer) {
      await this.waitForSignal();
    }
    this.readers++;
  }
  
  releaseRead(): void {
    this.readers--;
    if (this.readers === 0 && this.waitingWriters.length > 0) {
      this.waitingWriters.shift()!();
    }
  }
  
  async acquireWrite(): Promise<void> {
    while (this.writer || this.readers > 0) {
      await this.waitForSignal();
    }
    this.writer = true;
  }
  
  releaseWrite(): void {
    this.writer = false;
    if (this.waitingWriters.length > 0) {
      this.waitingWriters.shift()!();
    }
  }
  
  private waitForSignal(): Promise<void> {
    return new Promise(resolve => {
      this.waitingWriters.push(resolve);
    });
  }
}
```
