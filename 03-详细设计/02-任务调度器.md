# 任务调度器 (TaskScheduler) 详细设计

## 1. 概述

### 1.1 设计目标

任务调度器负责管理和调度所有Agent任务的执行，支持并发控制、任务优先级、暂停恢复等功能。

## 2. 数据结构

### 2.1 任务定义

```typescript
interface Task {
  id: string;
  type: TaskType;
  agent_type: AgentType;
  priority: number;           // 1-5，5最高
  status: TaskStatus;
  input: TaskInput;
  output?: TaskOutput;
  error?: TaskError;
  
  // 时间信息
  created_at: Date;
  started_at?: Date;
  completed_at?: Date;
  
  // 重试信息
  retry_count: number;
  max_retries: number;
  
  // 依赖信息
  dependencies: string[];     // 依赖的任务ID
  parent_id?: string;         // 父任务ID
  
  // 元数据
  metadata: {
    bid_id: string;
    chapter_id?: string;
    agent_id?: string;
    timeout?: number;
  };
}

enum TaskType {
  ANALYZE_TENDER = 'analyze_tender',
  GENERATE_OUTLINE = 'generate_outline',
  GENERATE_CONTENT = 'generate_content',
  REVIEW_CONTENT = 'review_content',
  POLISH_CONTENT = 'polish_content',
  EXPORT_BID = 'export_bid'
}

enum TaskStatus {
  PENDING = 'pending',
  QUEUED = 'queued',
  RUNNING = 'running',
  PAUSED = 'paused',
  COMPLETED = 'completed',
  FAILED = 'failed',
  CANCELLED = 'cancelled'
}

interface TaskInput {
  [key: string]: any;
}

interface TaskOutput {
  [key: string]: any;
}

interface TaskError {
  code: string;
  message: string;
  details?: any;
  stack?: string;
}
```

## 5. 实现代码

```typescript
import { EventEmitter } from 'events';
import { PriorityQueue } from '../utils/PriorityQueue';
import { AgentPool } from '../agents/AgentPool';
import { StateManager } from './StateManager';
import { Logger } from '../utils/Logger';

export class TaskScheduler extends EventEmitter implements ITaskScheduler {
  private queue: PriorityQueue<Task>;
  private runningTasks: Map<string, Task>;
  private completedTasks: Map<string, TaskResult>;
  private agentPool: AgentPool;
  private stateManager: StateManager;
  private logger: Logger;
  private config: SchedulerConfig;
  private workers: Worker[];
  private isRunning: boolean;
  
  constructor(
    agentPool: AgentPool,
    stateManager: StateManager,
    config: SchedulerConfig,
    logger: Logger
  ) {
    super();
    this.agentPool = agentPool;
    this.stateManager = stateManager;
    this.config = config;
    this.logger = logger;
    
    this.queue = new PriorityQueue<Task>((a, b) => b.priority - a.priority);
    this.runningTasks = new Map();
    this.completedTasks = new Map();
    this.workers = [];
    this.isRunning = false;
  }
  
  async start(): Promise<void> {
    if (this.isRunning) {
      return;
    }
    
    this.isRunning = true;
    
    // 创建Worker
    for (let i = 0; i < this.config.max_concurrency; i++) {
      const worker = new Worker(i, this);
      this.workers.push(worker);
      worker.start();
    }
    
    this.logger.info(`任务调度器已启动，并发数: ${this.config.max_concurrency}`);
  }
  
  async stop(): Promise<void> {
    if (!this.isRunning) {
      return;
    }
    
    this.isRunning = false;
    
    // 停止所有Worker
    for (const worker of this.workers) {
      await worker.stop();
    }
    
    this.logger.info('任务调度器已停止');
  }
  
  async submit(task: Task): Promise<TaskResult> {
    // 验证任务
    this.validateTask(task);
    
    // 设置初始状态
    task.status = TaskStatus.QUEUED;
    task.created_at = new Date();
    task.retry_count = 0;
    
    // 加入队列
    this.queue.enqueue(task);
    
    this.emit('task_submitted', task);
    this.logger.info(`任务已提交: ${task.id}, 类型: ${task.type}`);
    
    // 返回Promise，任务完成时resolve
    return new Promise((resolve, reject) => {
      const onComplete = (result: TaskResult) => {
        if (result.task_id === task.id) {
          this.removeListener('task_completed', onComplete);
          this.removeListener('task_failed', onFailed);
          resolve(result);
        }
      };
      
      const onFailed = (result: TaskResult) => {
        if (result.task_id === task.id) {
          this.removeListener('task_completed', onComplete);
          this.removeListener('task_failed', onFailed);
          reject(result.error);
        }
      };
      
      this.on('task_completed', onComplete);
      this.on('task_failed', onFailed);
    });
  }
  
  async submitBatch(tasks: Task[]): Promise<TaskResult[]> {
    const promises = tasks.map(task => this.submit(task));
    return Promise.all(promises);
  }
  
  async cancel(taskId: string): Promise<void> {
    const task = this.getTask(taskId);
    
    if (!task) {
      throw new Error(`Task not found: ${taskId}`);
    }
    
    if (task.status === TaskStatus.COMPLETED || task.status === TaskStatus.CANCELLED) {
      return;
    }
    
    // 从队列中移除
    this.queue.remove(task);
    
    // 如果正在运行，停止Agent
    if (task.status === TaskStatus.RUNNING && task.metadata.agent_id) {
      await this.agentPool.release(task.metadata.agent_id);
    }
    
    task.status = TaskStatus.CANCELLED;
    task.completed_at = new Date();
    
    this.runningTasks.delete(taskId);
    this.emit('task_cancelled', task);
    
    this.logger.info(`任务已取消: ${taskId}`);
  }
  
  async pause(taskId: string): Promise<void> {
    const task = this.getTask(taskId);
    
    if (!task || task.status !== TaskStatus.RUNNING) {
      throw new Error(`Cannot pause task: ${taskId}`);
    }
    
    task.status = TaskStatus.PAUSED;
    
    // 保存状态
    await this.stateManager.saveState(taskId, task);
    
    this.emit('task_paused', task);
    this.logger.info(`任务已暂停: ${taskId}`);
  }
  
  async resume(taskId: string): Promise<void> {
    const task = this.getTask(taskId);
    
    if (!task || task.status !== TaskStatus.PAUSED) {
      throw new Error(`Cannot resume task: ${taskId}`);
    }
    
    // 恢复状态
    const savedState = await this.stateManager.restoreState(taskId);
    if (savedState) {
      Object.assign(task, savedState);
    }
    
    task.status = TaskStatus.QUEUED;
    this.queue.enqueue(task);
    
    this.emit('task_resumed', task);
    this.logger.info(`任务已恢复: ${taskId}`);
  }
  
  getTask(taskId: string): Task | undefined {
    return this.runningTasks.get(taskId) || 
           this.queue.find(t => t.id === taskId) ||
           this.completedTasks.get(taskId)?.task;
  }
  
  getTasksByStatus(status: TaskStatus): Task[] {
    const tasks: Task[] = [];
    
    if (status === TaskStatus.QUEUED) {
      tasks.push(...this.queue.toArray());
    } else if (status === TaskStatus.RUNNING) {
      tasks.push(...this.runningTasks.values());
    }
    
    return tasks;
  }
  
  getTasksByBid(bidId: string): Task[] {
    const tasks: Task[] = [];
    
    for (const task of this.queue.toArray()) {
      if (task.metadata.bid_id === bidId) {
        tasks.push(task);
      }
    }
    
    for (const task of this.runningTasks.values()) {
      if (task.metadata.bid_id === bidId) {
        tasks.push(task);
      }
    }
    
    return tasks;
  }
  
  // Worker调用：获取下一个任务
  async getNextTask(): Promise<Task | null> {
    if (this.queue.isEmpty()) {
      return null;
    }
    
    const task = this.queue.dequeue();
    
    // 检查依赖
    if (!this.areDependenciesMet(task)) {
      this.queue.enqueue(task);
      return null;
    }
    
    // 检查Agent可用性
    const agent = await this.agentPool.acquire(task.agent_type);
    if (!agent) {
      this.queue.enqueue(task);
      return null;
    }
    
    task.status = TaskStatus.RUNNING;
    task.started_at = new Date();
    task.metadata.agent_id = agent.id;
    
    this.runningTasks.set(task.id, task);
    
    this.emit('task_started', task);
    this.logger.info(`任务开始执行: ${task.id}`);
    
    return task;
  }
  
  // Worker调用：任务完成
  async completeTask(taskId: string, output: TaskOutput): Promise<void> {
    const task = this.runningTasks.get(taskId);
    
    if (!task) {
      return;
    }
    
    task.status = TaskStatus.COMPLETED;
    task.completed_at = new Date();
    task.output = output;
    
    const result: TaskResult = {
      task_id: taskId,
      status: TaskStatus.COMPLETED,
      output,
      duration: task.completed_at.getTime() - task.started_at!.getTime(),
      retry_count: task.retry_count
    };
    
    this.runningTasks.delete(taskId);
    this.completedTasks.set(taskId, result);
    
    // 释放Agent
    if (task.metadata.agent_id) {
      await this.agentPool.release(task.metadata.agent_id);
    }
    
    this.emit('task_completed', result);
    this.logger.info(`任务完成: ${taskId}, 耗时: ${result.duration}ms`);
  }
  
  // Worker调用：任务失败
  async failTask(taskId: string, error: TaskError): Promise<void> {
    const task = this.runningTasks.get(taskId);
    
    if (!task) {
      return;
    }
    
    task.retry_count++;
    
    // 检查是否可以重试
    if (task.retry_count < task.max_retries) {
      task.status = TaskStatus.QUEUED;
      this.queue.enqueue(task);
      
      this.emit('task_retrying', task);
      this.logger.warn(`任务重试: ${taskId}, 第${task.retry_count}次`);
    } else {
      task.status = TaskStatus.FAILED;
      task.completed_at = new Date();
      task.error = error;
      
      const result: TaskResult = {
        task_id: taskId,
        status: TaskStatus.FAILED,
        error,
        duration: task.completed_at.getTime() - task.started_at!.getTime(),
        retry_count: task.retry_count
      };
      
      this.completedTasks.set(taskId, result);
      
      this.emit('task_failed', result);
      this.logger.error(`任务失败: ${taskId}, 错误: ${error.message}`);
    }
    
    this.runningTasks.delete(taskId);
    
    // 释放Agent
    if (task.metadata.agent_id) {
      await this.agentPool.release(task.metadata.agent_id);
    }
  }
  
  private validateTask(task: Task): void {
    if (!task.id || !task.type || !task.agent_type) {
      throw new Error('Invalid task: missing required fields');
    }
    
    if (task.priority < 1 || task.priority > 5) {
      throw new Error('Invalid task priority: must be 1-5');
    }
  }
  
  private areDependenciesMet(task: Task): boolean {
    for (const depId of task.dependencies) {
      const depResult = this.completedTasks.get(depId);
      if (!depResult || depResult.status !== TaskStatus.COMPLETED) {
        return false;
      }
    }
    return true;
  }
}

class Worker {
  private id: number;
  private scheduler: TaskScheduler;
  private isRunning: boolean;
  private currentTask: Task | null;
  
  constructor(id: number, scheduler: TaskScheduler) {
    this.id = id;
    this.scheduler = scheduler;
    this.isRunning = false;
    this.currentTask = null;
  }
  
  async start(): Promise<void> {
    this.isRunning = true;
    this.run();
  }
  
  async stop(): Promise<void> {
    this.isRunning = false;
  }
  
  private async run(): Promise<void> {
    while (this.isRunning) {
      try {
        const task = await this.scheduler.getNextTask();
        
        if (!task) {
          await this.sleep(100);
          continue;
        }
        
        this.currentTask = task;
        
        // 执行任务
        const agent = await this.scheduler.agentPool.getAgent(task.metadata.agent_id!);
        const output = await agent.execute(task.input);
        
        await this.scheduler.completeTask(task.id, output);
      } catch (error) {
        if (this.currentTask) {
          await this.scheduler.failTask(this.currentTask.id, {
            code: 'EXECUTION_ERROR',
            message: error.message,
            stack: error.stack
          });
        }
      } finally {
        this.currentTask = null;
      }
    }
  }
  
  private sleep(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

## 6. 分组调度策略

### 6.1 分组串行调度

```typescript
class GroupScheduler {
  private scheduler: TaskScheduler;
  
  async scheduleGroup(group: ChapterGroup): Promise<ChapterContent[]> {
    const results: ChapterContent[] = [];
    
    // 组内串行执行
    for (const chapter of group.chapters) {
      const task: Task = {
        id: `task-${chapter.id}`,
        type: TaskType.GENERATE_CONTENT,
        agent_type: AgentType.WRITER,
        priority: 3,
        status: TaskStatus.PENDING,
        input: { chapter },
        retry_count: 0,
        max_retries: 3,
        dependencies: results.length > 0 ? [`task-${results[results.length - 1].chapter_id}`] : [],
        metadata: {
          bid_id: group.bidId,
          chapter_id: chapter.id
        }
      };
      
      const result = await this.scheduler.submit(task);
      results.push(result.output);
    }
    
    return results;
  }
  
  async scheduleGroupsParallel(groups: ChapterGroup[]): Promise<ChapterContent[][]> {
    // 跨组并发执行
    const promises = groups.map(group => this.scheduleGroup(group));
    return Promise.all(promises);
  }
}
```
