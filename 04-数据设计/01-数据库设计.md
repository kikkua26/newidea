# 数据库设计

## 1. 设计概述

### 1.1 设计目标

- 支持标书项目的完整生命周期管理
- 支持多Agent协作的数据共享
- 支持任务状态的持久化和恢复
- 支持知识库的管理

### 1.2 技术选型

| 数据库 | 用途 | 说明 |
|--------|------|------|
| **SQLite** | 结构化数据 | 轻量、无需额外服务 |
| **文件系统** | 文件存储 | 招标文件、导出文件 |

## 2. 数据库Schema

### 2.1 完整DDL

```sql
-- ============================================
-- 标书项目表
-- ============================================
CREATE TABLE IF NOT EXISTS bids (
  id TEXT PRIMARY KEY,
  project_name TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'created',
  config JSON,
  
  -- 关联ID
  tender_doc_id TEXT,
  analysis_id TEXT,
  outline_id TEXT,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  completed_at DATETIME,
  
  -- 元数据
  metadata JSON,
  
  FOREIGN KEY (tender_doc_id) REFERENCES tender_documents(id),
  FOREIGN KEY (analysis_id) REFERENCES analyses(id),
  FOREIGN KEY (outline_id) REFERENCES outlines(id)
);

-- 索引
CREATE INDEX idx_bids_status ON bids(status);
CREATE INDEX idx_bids_created_at ON bids(created_at);

-- ============================================
-- 招标文件表
-- ============================================
CREATE TABLE IF NOT EXISTS tender_documents (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  
  -- 文件信息
  file_name TEXT NOT NULL,
  file_type TEXT NOT NULL,  -- pdf, word, image
  file_path TEXT NOT NULL,
  file_size INTEGER,
  
  -- 内容
  content TEXT,             -- 提取的文本内容
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE
);

-- ============================================
-- 分析结果表
-- ============================================
CREATE TABLE IF NOT EXISTS analyses (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  
  -- 项目信息
  project_name TEXT,
  project_overview TEXT,
  project_location TEXT,
  project_budget TEXT,
  project_duration TEXT,
  
  -- 结构化数据
  technical_requirements JSON,
  scoring_criteria JSON,
  qualification_requirements JSON,
  business_conditions JSON,
  key_points JSON,
  
  -- 元数据
  metadata JSON,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE
);

-- ============================================
-- 大纲表
-- ============================================
CREATE TABLE IF NOT EXISTS outlines (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  
  -- 大纲内容
  items JSON NOT NULL,
  version INTEGER DEFAULT 1,
  
  -- 统计信息
  total_chapters INTEGER DEFAULT 0,
  max_depth INTEGER DEFAULT 0,
  requirement_coverage REAL DEFAULT 0,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE
);

-- ============================================
-- 章节表
-- ============================================
CREATE TABLE IF NOT EXISTS chapters (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  outline_item_id TEXT NOT NULL,
  
  -- 章节信息
  title TEXT NOT NULL,
  content TEXT,
  level INTEGER NOT NULL,
  sort_order INTEGER NOT NULL,
  
  -- 状态
  status TEXT DEFAULT 'pending',  -- pending, generating, generated, reviewing, reviewed, polishing, polished, error
  
  -- 元数据
  metadata JSON,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE
);

-- 索引
CREATE INDEX idx_chapters_bid_id ON chapters(bid_id);
CREATE INDEX idx_chapters_status ON chapters(status);

-- ============================================
-- 编排决策表
-- ============================================
CREATE TABLE IF NOT EXISTS content_plans (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  chapter_id TEXT NOT NULL,
  
  -- 编排决策
  plan JSON NOT NULL,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE,
  FOREIGN KEY (chapter_id) REFERENCES chapters(id) ON DELETE CASCADE
);

-- ============================================
-- 审核报告表
-- ============================================
CREATE TABLE IF NOT EXISTS reviews (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  chapter_id TEXT,
  
  -- 审核结果
  overall_score INTEGER,
  passed BOOLEAN DEFAULT FALSE,
  dimensions JSON,
  issues JSON,
  suggestions JSON,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE,
  FOREIGN KEY (chapter_id) REFERENCES chapters(id) ON DELETE CASCADE
);

-- ============================================
-- 润色记录表
-- ============================================
CREATE TABLE IF NOT EXISTS polish_records (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  chapter_id TEXT NOT NULL,
  
  -- 润色内容
  original_content TEXT,
  polished_content TEXT,
  changes JSON,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE,
  FOREIGN KEY (chapter_id) REFERENCES chapters(id) ON DELETE CASCADE
);

-- ============================================
-- 术语表
-- ============================================
CREATE TABLE IF NOT EXISTS terminology (
  id TEXT PRIMARY KEY,
  bid_id TEXT,              -- NULL表示全局术语
  
  -- 术语信息
  term TEXT NOT NULL,
  definition TEXT NOT NULL,
  aliases JSON,
  
  -- 使用统计
  usage_count INTEGER DEFAULT 0,
  first_used_in TEXT,       -- 首次使用的章节ID
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE
);

-- 索引
CREATE INDEX idx_terminology_term ON terminology(term);
CREATE INDEX idx_terminology_bid_id ON terminology(bid_id);

-- ============================================
-- 章节摘要表
-- ============================================
CREATE TABLE IF NOT EXISTS chapter_summaries (
  id TEXT PRIMARY KEY,
  chapter_id TEXT NOT NULL,
  
  -- 摘要信息
  summary TEXT NOT NULL,
  key_points JSON,
  word_count INTEGER,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (chapter_id) REFERENCES chapters(id) ON DELETE CASCADE
);

-- ============================================
-- 交叉引用表
-- ============================================
CREATE TABLE IF NOT EXISTS references (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  
  -- 引用关系
  from_chapter_id TEXT NOT NULL,
  to_chapter_id TEXT NOT NULL,
  context TEXT,
  ref_type TEXT,            -- see_also, as_shown_in, refer_to, mentioned_in
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE,
  FOREIGN KEY (from_chapter_id) REFERENCES chapters(id) ON DELETE CASCADE,
  FOREIGN KEY (to_chapter_id) REFERENCES chapters(id) ON DELETE CASCADE
);

-- ============================================
-- 评分点覆盖表
-- ============================================
CREATE TABLE IF NOT EXISTS requirement_coverage (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  
  -- 评分点信息
  requirement_id TEXT NOT NULL,
  requirement_desc TEXT,
  
  -- 覆盖信息
  covered_by JSON,          -- 覆盖的章节ID列表
  coverage_status TEXT,     -- full, partial, none
  
  -- 时间戳
  last_updated DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE
);

-- ============================================
-- 任务表
-- ============================================
CREATE TABLE IF NOT EXISTS tasks (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  
  -- 任务信息
  type TEXT NOT NULL,
  agent_type TEXT NOT NULL,
  priority INTEGER DEFAULT 3,
  status TEXT DEFAULT 'pending',
  
  -- 输入输出
  input JSON,
  output JSON,
  error JSON,
  
  -- 重试信息
  retry_count INTEGER DEFAULT 0,
  max_retries INTEGER DEFAULT 3,
  
  -- 依赖
  dependencies JSON,
  parent_id TEXT,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  started_at DATETIME,
  completed_at DATETIME,
  
  -- 元数据
  metadata JSON,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE
);

-- 索引
CREATE INDEX idx_tasks_bid_id ON tasks(bid_id);
CREATE INDEX idx_tasks_status ON tasks(status);
CREATE INDEX idx_tasks_type ON tasks(type);

-- ============================================
-- 共享上下文快照表
-- ============================================
CREATE TABLE IF NOT EXISTS shared_contexts (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  
  -- 上下文数据
  data JSON NOT NULL,
  version INTEGER DEFAULT 1,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (bid_id) REFERENCES bids(id) ON DELETE CASCADE
);

-- ============================================
-- 知识库表
-- ============================================
CREATE TABLE IF NOT EXISTS knowledge_bases (
  id TEXT PRIMARY KEY,
  
  -- 知识库信息
  name TEXT NOT NULL,
  description TEXT,
  type TEXT,                -- technical, company, project
  
  -- 统计信息
  item_count INTEGER DEFAULT 0,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- ============================================
-- 知识条目表
-- ============================================
CREATE TABLE IF NOT EXISTS knowledge_items (
  id TEXT PRIMARY KEY,
  knowledge_base_id TEXT NOT NULL,
  
  -- 条目信息
  title TEXT NOT NULL,
  content TEXT NOT NULL,
  category TEXT,
  tags JSON,
  
  -- 向量嵌入（可选）
  embedding JSON,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (knowledge_base_id) REFERENCES knowledge_bases(id) ON DELETE CASCADE
);

-- 索引
CREATE INDEX idx_knowledge_items_kb_id ON knowledge_items(knowledge_base_id);
CREATE INDEX idx_knowledge_items_category ON knowledge_items(category);

-- ============================================
-- 操作日志表
-- ============================================
CREATE TABLE IF NOT EXISTS operation_logs (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  
  -- 操作信息
  operation TEXT NOT NULL,
  entity_type TEXT NOT NULL,
  entity_id TEXT,
  
  -- 操作详情
  details JSON,
  
  -- 时间戳
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 索引
CREATE INDEX idx_operation_logs_entity ON operation_logs(entity_type, entity_id);
CREATE INDEX idx_operation_logs_created_at ON operation_logs(created_at);
```

## 4. 数据访问层设计

### 4.1 Repository模式

```typescript
// 基础Repository接口
interface IRepository<T> {
  findById(id: string): Promise<T | null>;
  findAll(filter?: Filter): Promise<T[]>;
  create(entity: T): Promise<T>;
  update(id: string, entity: Partial<T>): Promise<T>;
  delete(id: string): Promise<void>;
}

// 标书Repository
class BidRepository implements IRepository<Bid> {
  async findById(id: string): Promise<Bid | null> {
    const row = await db.get('SELECT * FROM bids WHERE id = ?', [id]);
    return row ? this.mapToEntity(row) : null;
  }
  
  async create(bid: Bid): Promise<Bid> {
    await db.run(
      'INSERT INTO bids (id, project_name, status, config) VALUES (?, ?, ?, ?)',
      [bid.id, bid.project_name, bid.status, JSON.stringify(bid.config)]
    );
    return bid;
  }
  
  async updateStatus(id: string, status: BidStatus): Promise<void> {
    await db.run(
      'UPDATE bids SET status = ?, updated_at = CURRENT_TIMESTAMP WHERE id = ?',
      [status, id]
    );
  }
}
```

### 4.2 数据库连接管理

```typescript
class DatabaseManager {
  private db: Database;
  
  async initialize(dbPath: string): Promise<void> {
    this.db = await open({
      filename: dbPath,
      driver: sqlite3.Database
    });
    
    // 启用WAL模式
    await this.db.run('PRAGMA journal_mode=WAL');
    
    // 创建表
    await this.createTables();
  }
  
  async close(): Promise<void> {
    await this.db.close();
  }
  
  getDatabase(): Database {
    return this.db;
  }
}
```
