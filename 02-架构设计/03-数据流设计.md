# 数据流设计

## 1. 整体数据流

### 1.1 主流程数据流

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                              标书生成主流程数据流                             │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│  用户输入                                                                   │
│      │                                                                      │
│      ↓                                                                      │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │ ① 招标文件上传                                                      │   │
│  │    输入: PDF/Word/图片文件                                           │   │
│  │    输出: TenderDocument { id, fileName, content, fileType }          │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│      │                                                                      │
│      ↓                                                                      │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │ ② 招标文件分析 (Analyst Agent)                                      │   │
│  │    输入: TenderDocument.content                                      │   │
│  │    输出: TenderAnalysis {                                            │   │
│  │      project_name, project_overview,                                │   │
│  │      technical_requirements[], scoring_criteria[],                   │   │
│  │      qualification_requirements[], key_points[]                      │   │
│  │    }                                                                 │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│      │                                                                      │
│      ↓                                                                      │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │ ③ 大纲生成 (Outliner Agent)                                         │   │
│  │    输入: TenderAnalysis                                              │   │
│  │    输出: Outline {                                                   │   │
│  │      id, title, description, level,                                 │   │
│  │      requirement_ids[], children[]                                   │   │
│  │    }                                                                 │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│      │                                                                      │
│      ↓                                                                      │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │ ④ 用户确认/调整大纲                                                  │   │
│  │    输入: Outline                                                     │   │
│  │    输出: ConfirmedOutline (用户确认后的Outline)                      │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│      │                                                                      │
│      ↓                                                                      │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │ ⑤ 编排决策 (Orchestrator)                                           │   │
│  │    输入: ConfirmedOutline, 配置选项                                  │   │
│  │    输出: ContentPlan[] {                                             │   │
│  │      chapterId, useTable, useImage, useMermaid,                     │   │
│  │      knowledgeIds[]                                                  │   │
│  │    }                                                                 │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│      │                                                                      │
│      ↓                                                                      │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │ ⑥ 分组串行+跨组并发生成 (Writer Agents)                              │   │
│  │    输入: ConfirmedOutline, ContentPlan[], SharedContext               │   │
│  │    输出: Chapter[] {                                                 │   │
│  │      id, title, content, status, metadata                           │   │
│  │    }                                                                 │   │
│  │                                                                      │   │
│  │    并发策略:                                                          │   │
│  │    ┌─────────────────────────────────────────────────────────────┐  │   │
│  │    │ Group 1 (串行): 1.1 → 1.2 → 1.3                           │  │   │
│  │    │ Group 2 (串行): 2.1 → 2.2 → 2.3                           │  │   │
│  │    │ Group 3 (串行): 3.1 → 3.2 → 3.3                           │  │   │
│  │    │                                                             │  │   │
│  │    │ 跨组并发: Group 1 ∥ Group 2 ∥ Group 3                      │  │   │
│  │    └─────────────────────────────────────────────────────────────┘  │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│      │                                                                      │
│      ↓                                                                      │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │ ⑦ 审核 (Reviewer Agent)                                             │   │
│  │    输入: Chapter[], Outline                                          │   │
│  │    输出: ReviewReport {                                              │   │
│  │      score, issues[], suggestions[],                                 │   │
│  │      passed: boolean                                                 │   │
│  │    }                                                                 │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│      │                                                                      │
│      ├──→ passed=false ──→ 返回⑥修正                                     │
│      │                                                                      │
│      ↓ passed=true                                                         │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │ ⑧ 润色 (Polisher Agent)                                             │   │
│  │    输入: Chapter[], ReviewReport                                     │   │
│  │    输出: PolishedChapter[] {                                         │   │
│  │      id, title, polished_content, metadata                          │   │
│  │    }                                                                 │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│      │                                                                      │
│      ↓                                                                      │
│  ┌─────────────────────────────────────────────────────────────────────┐   │
│  │ ⑨ 导出                                                              │   │
│  │    输入: PolishedChapter[], ExportConfig                             │   │
│  │    输出: BidDocument (Word/PDF/Markdown)                             │   │
│  └─────────────────────────────────────────────────────────────────────┘   │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘
```

## 2. 共享上下文数据流

### 2.1 上下文更新流程

```
┌─────────────────────────────────────────────────────────────────────────────┐
│                           共享上下文更新流程                                 │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                             │
│   Writer Agent 完成章节                                                      │
│         │                                                                   │
│         ↓                                                                   │
│   ┌─────────────────────────────────────────────────────────────────────┐  │
│   │ ① 生成章节摘要                                                      │  │
│   │    输入: 章节内容                                                    │  │
│   │    输出: ChapterSummary {                                            │  │
│   │      chapter_id, summary (200字以内),                                │  │
│   │      key_points[], generated_at                                      │  │
│   │    }                                                                 │  │
│   └─────────────────────────────────────────────────────────────────────┘  │
│         │                                                                   │
│         ↓                                                                   │
│   ┌─────────────────────────────────────────────────────────────────────┐  │
│   │ ② 提取术语                                                          │  │
│   │    输入: 章节内容                                                    │  │
│   │    输出: Term[] {                                                    │  │
│   │      name, definition, aliases[], usage_count                        │  │
│   │    }                                                                 │  │
│   └─────────────────────────────────────────────────────────────────────┘  │
│         │                                                                   │
│         ↓                                                                   │
│   ┌─────────────────────────────────────────────────────────────────────┐  │
│   │ ③ 提取交叉引用                                                      │  │
│   │    输入: 章节内容, 其他章节标题                                       │  │
│   │    输出: Reference[] {                                               │  │
│   │      from_chapter, to_chapter, context, type                         │  │
│   │    }                                                                 │  │
│   └─────────────────────────────────────────────────────────────────────┘  │
│         │                                                                   │
│         ↓                                                                   │
│   ┌─────────────────────────────────────────────────────────────────────┐  │
│   │ ④ 更新评分点覆盖                                                    │  │
│   │    输入: 章节ID, 章节内容                                            │  │
│   │    输出: RequirementCoverage[] {                                     │  │
│   │      requirement_id, covered_by[], coverage_status                   │  │
│   │    }                                                                 │  │
│   └─────────────────────────────────────────────────────────────────────┘  │
│         │                                                                   │
│         ↓                                                                   │
│   ┌─────────────────────────────────────────────────────────────────────┐  │
│   │ ⑤ 写入共享上下文池                                                  │  │
│   │    SharedContext.terminology += newTerms                             │  │
│   │    SharedContext.chapterSummaries.set(id, summary)                   │  │
│   │    SharedContext.references += newRefs                               │  │
│   │    SharedContext.requirementCoverage.update(...)                      │  │
│   └─────────────────────────────────────────────────────────────────────┘  │
│         │                                                                   │
│         ↓                                                                   │
│   ┌─────────────────────────────────────────────────────────────────────┐  │
│   │ ⑥ 广播事件                                                          │  │
│   │    EventBus.emit('CHAPTER_COMPLETED', {                              │  │
│   │      chapterId, summary, newTerms, newRefs                          │  │
│   │    })                                                                │  │
│   └─────────────────────────────────────────────────────────────────────┘  │
│         │                                                                   │
│         ↓                                                                   │
│   ┌─────────────────────────────────────────────────────────────────────┐  │
│   │ ⑦ 其他写手接收更新                                                  │  │
│   │    Writer Agent.on('CHAPTER_COMPLETED', updateLocalContext)          │  │
│   └─────────────────────────────────────────────────────────────────────┘  │
│                                                                             │
└─────────────────────────────────────────────────────────────────────────────┘
```

## 3. 数据模型

### 3.1 核心数据模型

```typescript
// 招标文件
interface TenderDocument {
  id: string;
  file_name: string;
  file_type: 'pdf' | 'word' | 'image';
  content: string;           // 提取的文本内容
  raw_file_path: string;     // 原始文件路径
  created_at: Date;
}

// 招标分析结果
interface TenderAnalysis {
  id: string;
  tender_doc_id: string;
  project_name: string;
  project_overview: string;
  technical_requirements: string[];
  scoring_criteria: ScoringCategory[];
  qualification_requirements: string[];
  key_points: string[];
  created_at: Date;
}

// 评分大类
interface ScoringCategory {
  category: string;
  items: ScoringItem[];
}

// 评分细项
interface ScoringItem {
  item: string;
  score: number;
  description: string;
}

// 大纲
interface Outline {
  id: string;
  bid_id: string;
  items: OutlineItem[];
  version: number;
  created_at: Date;
  updated_at: Date;
}

// 大纲节点
interface OutlineItem {
  id: string;
  parent_id?: string;
  title: string;
  description: string;
  level: number;
  order: number;
  requirement_ids: string[];
  children?: OutlineItem[];
}

// 章节
interface Chapter {
  id: string;
  bid_id: string;
  outline_item_id: string;
  title: string;
  content: string;
  level: number;
  order: number;
  status: ChapterStatus;
  metadata: ChapterMetadata;
  created_at: Date;
  updated_at: Date;
}

// 章节状态
enum ChapterStatus {
  PENDING = 'pending',
  GENERATING = 'generating',
  GENERATED = 'generated',
  REVIEWING = 'reviewing',
  REVIEWED = 'reviewed',
  POLISHING = 'polishing',
  POLISHED = 'polished',
  ERROR = 'error'
}

// 章节元数据
interface ChapterMetadata {
  word_count: number;
  generation_time: number;
  review_score?: number;
  polish_count: number;
  knowledge_ids: string[];
  terminology_used: string[];
}

// 编排决策
interface ContentPlan {
  id: string;
  bid_id: string;
  chapter_id: string;
  use_table: boolean;
  table_purpose?: string;
  use_image: boolean;
  image_style?: string;
  image_prompt?: string;
  use_mermaid: boolean;
  mermaid_code?: string;
  knowledge_ids: string[];
  created_at: Date;
}

// 审核报告
interface ReviewReport {
  id: string;
  bid_id: string;
  chapter_id: string;
  score: number;
  issues: ReviewIssue[];
  suggestions: string[];
  passed: boolean;
  reviewer_id: string;
  created_at: Date;
}

// 审核问题
interface ReviewIssue {
  type: 'missing' | 'inconsistent' | 'format' | 'language' | 'logic';
  severity: 'high' | 'medium' | 'low';
  description: string;
  location?: string;
  suggestion: string;
}

// 标书项目
interface Bid {
  id: string;
  project_name: string;
  tender_doc_id: string;
  analysis_id: string;
  outline_id: string;
  status: BidStatus;
  config: BidConfig;
  created_at: Date;
  updated_at: Date;
}

// 标书状态
enum BidStatus {
  CREATED = 'created',
  ANALYZING = 'analyzing',
  ANALYZED = 'analyzed',
  OUTLINING = 'outlining',
  OUTLINED = 'outlined',
  GENERATING = 'generating',
  REVIEWING = 'reviewing',
  POLISHING = 'polishing',
  COMPLETED = 'completed',
  ERROR = 'error'
}

// 标书配置
interface BidConfig {
  ai_model: string;
  temperature: number;
  max_words: number;
  use_knowledge_base: boolean;
  knowledge_ids: string[];
  generation_concurrency: number;
  table_requirement: 'none' | 'light' | 'moderate' | 'heavy';
  use_ai_images: boolean;
  use_mermaid: boolean;
  export_format: 'word' | 'pdf' | 'markdown';
}
```

### 4.2 SQLite数据库设计

```sql
-- 项目表
CREATE TABLE bids (
  id TEXT PRIMARY KEY,
  project_name TEXT NOT NULL,
  tender_doc_id TEXT,
  analysis_id TEXT,
  outline_id TEXT,
  status TEXT DEFAULT 'created',
  config JSON,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 招标文件表
CREATE TABLE tender_documents (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  file_name TEXT NOT NULL,
  file_type TEXT NOT NULL,
  content TEXT,
  raw_file_path TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (bid_id) REFERENCES bids(id)
);

-- 分析结果表
CREATE TABLE analyses (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  project_name TEXT,
  project_overview TEXT,
  technical_requirements JSON,
  scoring_criteria JSON,
  qualification_requirements JSON,
  key_points JSON,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (bid_id) REFERENCES bids(id)
);

-- 大纲表
CREATE TABLE outlines (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  items JSON NOT NULL,
  version INTEGER DEFAULT 1,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (bid_id) REFERENCES bids(id)
);

-- 章节表
CREATE TABLE chapters (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  outline_item_id TEXT NOT NULL,
  title TEXT NOT NULL,
  content TEXT,
  level INTEGER,
  sort_order INTEGER,
  status TEXT DEFAULT 'pending',
  metadata JSON,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (bid_id) REFERENCES bids(id)
);

-- 编排决策表
CREATE TABLE content_plans (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  chapter_id TEXT NOT NULL,
  plan JSON NOT NULL,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (bid_id) REFERENCES bids(id),
  FOREIGN KEY (chapter_id) REFERENCES chapters(id)
);

-- 审核报告表
CREATE TABLE reviews (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  chapter_id TEXT NOT NULL,
  score INTEGER,
  issues JSON,
  suggestions JSON,
  passed BOOLEAN DEFAULT FALSE,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (bid_id) REFERENCES bids(id),
  FOREIGN KEY (chapter_id) REFERENCES chapters(id)
);

-- 术语表
CREATE TABLE terminology (
  id TEXT PRIMARY KEY,
  bid_id TEXT,
  term TEXT NOT NULL,
  definition TEXT,
  aliases JSON,
  usage_count INTEGER DEFAULT 0,
  is_global BOOLEAN DEFAULT FALSE,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (bid_id) REFERENCES bids(id)
);

-- 章节摘要表
CREATE TABLE chapter_summaries (
  id TEXT PRIMARY KEY,
  chapter_id TEXT NOT NULL,
  summary TEXT NOT NULL,
  key_points JSON,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (chapter_id) REFERENCES chapters(id)
);

-- 交叉引用表
CREATE TABLE references (
  id TEXT PRIMARY KEY,
  bid_id TEXT NOT NULL,
  from_chapter_id TEXT NOT NULL,
  to_chapter_id TEXT NOT NULL,
  context TEXT,
  ref_type TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (bid_id) REFERENCES bids(id),
  FOREIGN KEY (from_chapter_id) REFERENCES chapters(id),
  FOREIGN KEY (to_chapter_id) REFERENCES chapters(id)
);
```
